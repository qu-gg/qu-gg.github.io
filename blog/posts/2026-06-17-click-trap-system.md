### CLICK! Traps
These Traps are designed to be run within Adventure Sites, however they can find use during Journeying Encounters.

**The mechanics are straightforward:** when a player unknowingly triggers a Trap, you announce CLICK! and all players freeze. Go around the table, and see what their reaction is. Most reactions will fall into 3 categories: **Prone**, **Jump**, or **Brace**. Depending on what specific trap it is, the chosen reaction might incur an Edge or a Snag on the resulting Check to save. Failed Checks either directly apply the Failure Result or require a follow-up roll on the relevant Result Table.

Iterate and make the table your own! It is easy to reflavor or add traps, reward creativity, and factor in BREAK!!'s rules like Exploration for additional bonuses. Some examples of this could be: allowing Scouts to discover and disarm trap, automatically detecting which type of trap it is on Cautious Movement, factoring in PC Positions, and so on.

<script src="https://cdnjs.cloudflare.com/ajax/libs/html2canvas/1.4.1/html2canvas.min.js" crossorigin="anonymous" referrerpolicy="no-referrer"></script>
<script>
(function () {
  const figure = document.querySelector('.click-trap-figure');
  const btn = document.getElementById('click-trap-copy-btn');
  if (!figure || !btn) return;

  let cached = null;
  function render() {
    if (cached) return Promise.resolve(cached);
    if (typeof html2canvas === 'undefined') return Promise.reject(new Error('html2canvas not loaded'));
    return html2canvas(figure, { scale: 2, useCORS: true, backgroundColor: '#ffffff' })
      .then(canvas => { cached = canvas; return canvas; });
  }

  function flash(text, ok) {
    const original = 'Copy as Image to Clipboard';
    btn.textContent = text;
    if (ok) {
      btn.style.background = '#00b894';
      btn.style.borderColor = '#00b894';
      btn.style.color = '#fff';
    } else {
      btn.style.background = '#d63031';
      btn.style.borderColor = '#d63031';
      btn.style.color = '#fff';
    }
    setTimeout(() => {
      btn.textContent = original;
      btn.style.background = '';
      btn.style.borderColor = '';
      btn.style.color = '';
    }, 1800);
  }

  function copyImage() {
    render().then(canvas => {
      canvas.toBlob(blob => {
        if (!blob || !navigator.clipboard || !window.ClipboardItem) {
          flash('Clipboard unsupported', false);
          return;
        }
        navigator.clipboard.write([new ClipboardItem({ 'image/png': blob })])
          .then(() => flash('Copied!', true))
          .catch(() => flash('Copy failed', false));
      }, 'image/png');
    }).catch(() => flash('Render failed', false));
  }

  btn.addEventListener('click', copyImage);
  figure.addEventListener('contextmenu', e => {
    e.preventDefault();
    copyImage();
  });

  // Invalidate the cached render if the viewport size changes meaningfully.
  let lastWidth = window.innerWidth;
  window.addEventListener('resize', () => {
    if (Math.abs(window.innerWidth - lastWidth) > 40) {
      cached = null;
      lastWidth = window.innerWidth;
    }
  });
})();
</script>
